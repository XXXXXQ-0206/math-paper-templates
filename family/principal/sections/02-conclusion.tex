\section{Consequences}
\label{sec:conclusion}

\begin{corollary}
\label{cor:norm}
The functional $f \mapsto \int_X f \dd\mu$ is continuous on
$L^2(\mu)$ with operator norm at most $1$.
\end{corollary}

\begin{proof}
Immediate from \cref{eq:principal}.
\end{proof}
