\begin{theorem}[The supporting estimate]
\label{thm:support}
Let $f \in L^2(\mu)$ be bounded. Then
\begin{equation}
\label{eq:support}
  \norm{f}_{L^2(\mu)} \le \norm{f}_{L^\infty(\mu)} .
\end{equation}
\end{theorem}
