\begin{document}

\maketitle

% ---------------------------------------------------------------------
% Abstract: one paragraph, no citations, no cross-references.
% State (a) the problem, (b) the result, (c) why it is new.
% ---------------------------------------------------------------------
\begin{abstract}
This template reproduces the dominant manuscript conventions of the
openai/math preprint collection: an article class at 11\,pt, a modular
preamble, one shared theorem counter, numbered sections split across
the sections/ directory, and a BibTeX bibliography. Replace the text of
this abstract with a single self-contained paragraph stating the
problem, the main result, and its significance. Do not cite or
cross-reference anything here.
\end{abstract}
